Cadastro de Cardápio

// Carrega comércios do localStorage function loadCommerces() { const commerces = localStorage.getItem('commerces'); return commerces ? JSON.parse(commerces) : []; } // Carrega produtos do localStorage function loadProducts() { const products = localStorage.getItem('products'); return products ? JSON.parse(products) : []; } // Salva produtos no localStorage function saveProducts(products) { localStorage.setItem('products', JSON.stringify(products)); } // Preenche select de comércios function renderCommerceOptions() { const commerces = loadCommerces(); commerceSelect.innerHTML = ''; commerces.forEach((commerce, index) => { const option = document.createElement('option'); option.value = commerce.name; option.textContent = commerce.name; commerceSelect.appendChild(option); }); } // Renderiza lista de produtos do comércio selecionado function renderProducts() { const products = loadProducts(); const selectedCommerce = commerceSelect.value; const filtered = products.filter(p => p.commerce === selectedCommerce); productListDiv.innerHTML = ''; if (filtered.length === 0) { productListDiv.textContent = 'Nenhum produto cadastrado para este comércio.'; return; } filtered.forEach((product, index) => { const div = document.createElement('div'); div.className = 'list-item'; const infoDiv = document.createElement('div'); infoDiv.innerHTML = `${product.name} (${product.type}) - R$ ${product.price.toFixed(2)}
${product.description}`; const delBtn = document.createElement('button'); delBtn.textContent = 'Excluir'; delBtn.onclick = () => { if (confirm(`Excluir produto "${product.name}"?`)) { products.splice(products.indexOf(product), 1); saveProducts(products); renderProducts(); } }; div.appendChild(infoDiv); div.appendChild(delBtn); productListDiv.appendChild(div); }); } // Evento para mudar comércio selecionado commerceSelect.addEventListener('change', renderProducts); // Evento para adicionar produto productForm.addEventListener('submit', e => { e.preventDefault(); const name = document.getElementById('product-name').value.trim(); const type = document.getElementById('product-type').value; const description = document.getElementById('product-description').value.trim(); const price = parseFloat(document.getElementById('product-price').value); const commerce = commerceSelect.value; if (name && type && description && !isNaN(price) && commerce) { const products = loadProducts(); products.push({ name, type, description, price, commerce }); saveProducts(products); renderProducts(); productForm.reset(); } else { alert('Preencha todos os campos corretamente.'); } }); // Inicialização renderCommerceOptions(); if (commerceSelect.options.length > 0) { commerceSelect.selectedIndex = 0; renderProducts(); }
